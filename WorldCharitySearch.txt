    <script>
        function executeFreeWebSearch() {
            const query = document.getElementById('query').value;
            const chosenCause = document.getElementById('charity-cause').value;
            if (!query) return;

            console.log(`User query "${query}" processed. Routing metrics to: [${chosenCause}]`);

            const targetedSearchUrl = `https://duckduckgo.com{encodeURIComponent(query)}`;
            window.location.href = targetedSearchUrl;
        }

        document.getElementById('query').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') { executeFreeWebSearch(); }
        });
